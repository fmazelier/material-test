@Injectable({ providedIn: 'root' })
export class UsersService {
  private readonly http = inject(HttpClient);

  // STATE
  private readonly _users = signal<User[]>([]);
  private readonly _loading = signal(false);

  // SELECTORS
  readonly users = this._users.asReadonly();
  readonly loading = this._loading.asReadonly();

  readonly userCount = computed(() => this._users().length);

  // ACTIONS
  loadUsers() {
    this._loading.set(true);

    this.http.get<User[]>('/api/users').subscribe({
      next: (users) => this._users.set(users),
      error: () => this._loading.set(false),
      complete: () => this._loading.set(false),
    });
  }

  addUser(user: User) {
    this._users.update(list => [...list, user]);
  }
}

export class FileUploaderComponent implements ControlValueAccessor {
  @Input() accept = '*';
  @Input() multiple = false;
  @Input() maxFileSizeMB = 10; // 10Mo par défaut

  selectedFiles = signal<File[]>([]);
  isDragging = signal(false);

  // Pour le ControlValueAccessor
  onChange = (value: any) => {};
  onTouched = () => {};

  // --- Gestion du Drag & Drop ---
  onDragOver(event: DragEvent) {
    event.preventDefault();
    event.stopPropagation();
    this.isDragging.set(true);
  }

  onDragLeave(event: DragEvent) {
    event.preventDefault();
    event.stopPropagation();
    this.isDragging.set(false);
  }

  onDrop(event: DragEvent) {
    event.preventDefault();
    event.stopPropagation();
    this.isDragging.set(false);

    const files = event.dataTransfer?.files;
    if (files) this.processFiles(files);
  }

  // --- Logique métier ---
  onFileChange(event: any) {
    const files = event.target.files;
    this.processFiles(files);
  }

  private processFiles(fileList: FileList) {
    let files = Array.from(fileList);

    // Validation taille
    files = files.filter(f => f.size / 1024 / 1024 <= this.maxFileSizeMB);

    if (this.multiple) {
      const current = this.selectedFiles();
      this.selectedFiles.set([...current, ...files]);
    } else {
      this.selectedFiles.set(files.slice(0, 1));
    }

    this.notifyParent();
  }

  removeFile(file: File, event: MouseEvent) {
    event.stopPropagation();
    this.selectedFiles.update(files => files.filter(f => f !== file));
    this.notifyParent();
  }

  private notifyParent() {
    const value = this.multiple ? this.selectedFiles() : (this.selectedFiles()[0] || null);
    this.onChange(value);
    this.onTouched();
  }

  // --- Helpers ---
  isImage(file: File): boolean {
    return file.type.startsWith('image/');
  }

  getPreview(file: File): string {
    return URL.createObjectURL(file);
  }

  // --- Méthodes ControlValueAccessor ---
  writeValue(value: any): void {
    if (!value) {
      this.selectedFiles.set([]);
    } else {
      this.selectedFiles.set(Array.isArray(value) ? value : [value]);
    }
  }
  registerOnChange(fn: any): void { this.onChange = fn; }
  registerOnTouched(fn: any): void { this.onTouched = fn; }
}

import { Component, ElementRef, HostBinding, Input, OnDestroy, Optional, Self, ViewChild, computed, signal } from '@angular/core';
import { ControlValueAccessor, NgControl, Validators } from '@angular/forms';
import { MatFormFieldControl } from '@angular/material/form-field';
import { MatIconModule } from '@angular/material/icon';
import { Subject } from 'rxjs';
import { FocusMonitor } from '@angular/cdk/a11y';

@Component({
  selector: 'app-file-control',
  standalone: true,
  imports: [MatIconModule],
  providers: [{ provide: MatFormFieldControl, useExisting: FileFormControlComponent }],
  template: `
    <div
      class="flex items-center gap-2 w-full cursor-pointer min-h-[1.5rem]"
      (click)="fileInput.click()"
      (dragover)="onDragOver($event)"
      (dragleave)="onDragLeave($event)"
      (drop)="onDrop($event)"
      [class.opacity-50]="disabled()"
    >
      <input #fileInput type="file" class="hidden" [multiple]="multiple" [accept]="accept" (change)="onFileChange($event)">

      <div class="flex-1 truncate text-sm font-normal">
        @if (empty()) {
          <span class="text-gray-400 italic">{{ placeholder }}</span>
        } @else {
          <span class="text-gray-900">{{ displayText() }}</span>
        }
      </div>

      <mat-icon class="text-gray-400 scale-90">attach_file</mat-icon>
    </div>
  `,
  styles: `
    :host { display: block; }
  `
})
export class FileFormControlComponent implements ControlValueAccessor, MatFormFieldControl<File | File[]>, OnDestroy {
  // --- Propriétés obligatoires MatFormFieldControl ---
  static nextId = 0;
  @HostBinding() id = `file-control-${FileFormControlComponent.nextId++}`;

  stateChanges = new Subject<void>();
  focused = signal(false);
  touched = signal(false);
  disabled = signal(false);

  @Input() placeholder: string = '';
  @Input() multiple = false;
  @Input() accept = '*';
  @Input() required: boolean = false;

  // Valeur interne
  valueSignal = signal<File | File[] | null>(null);

  // Getters/Setters pour le form field
  get value() { return this.valueSignal(); }
  set value(val: File | File[] | null) {
    this.valueSignal.set(val);
    this.onChange(val);
    this.stateChanges.next();
  }

  // Est-ce que le champ est vide ? (Détermine si le label flotte)
  empty = computed(() => {
    const val = this.valueSignal();
    return Array.isArray(val) ? val.length === 0 : val === null;
  });

  @HostBinding('class.floating')
  get shouldLabelFloat() { return this.focused() || !this.empty(); }

  get errorState(): boolean {
    return !!(this.ngControl?.invalid && (this.ngControl?.touched || this.touched()));
  }

  controlType = 'app-file-control';

  // Texte à afficher dans le champ
  displayText = computed(() => {
    const val = this.valueSignal();
    if (!val) return '';
    if (Array.isArray(val)) return `${val.length} fichiers sélectionnés`;
    return val.name;
  });

  constructor(
    @Optional() @Self() public ngControl: NgControl,
    private fm: FocusMonitor,
    private elRef: ElementRef<HTMLElement>
  ) {
    if (this.ngControl) this.ngControl.valueAccessor = this;

    // Gère le focus pour l'accessibilité
    this.fm.monitor(elRef.nativeElement, true).subscribe(origin => {
      this.focused.set(!!origin);
      this.stateChanges.next();
    });
  }

  // --- Logique de fichier ---
  onFileChange(event: any) {
    const files = Array.from(event.target.files as FileList);
    this.value = this.multiple ? files : (files[0] || null);
    this.onTouched();
  }

  onDragOver(e: DragEvent) { e.preventDefault(); this.focused.set(true); }
  onDragLeave(e: DragEvent) { e.preventDefault(); this.focused.set(false); }
  onDrop(e: DragEvent) {
    e.preventDefault();
    const files = Array.from(e.dataTransfer?.files || []);
    this.value = this.multiple ? files : (files[0] || null);
    this.onTouched();
    this.focused.set(false);
  }

  // --- ControlValueAccessor ---
  onChange = (_: any) => {};
  onTouched = () => { this.touched.set(true); this.stateChanges.next(); };
  writeValue(value: any): void { this.valueSignal.set(value); this.stateChanges.next(); }
  registerOnChange(fn: any): void { this.onChange = fn; }
  registerOnTouched(fn: any): void { this.onTouched = fn; }
  setDisabledState(isDisabled: boolean): void { this.disabled.set(isDisabled); this.stateChanges.next(); }

  // --- Nettoyage ---
  ngOnDestroy() {
    this.stateChanges.complete();
    this.fm.stopMonitoring(this.elRef.nativeElement);
  }

  // Obligatoire pour MatFormFieldControl
  setDescribedByIds(ids: string[]) {}
  onContainerClick(event: MouseEvent) {
    if ((event.target as Element).tagName.toLowerCase() !== 'input') {
      this.elRef.nativeElement.querySelector('input')?.click();
    }
  }
}
